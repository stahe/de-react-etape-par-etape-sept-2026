// vite.config.ts - die Konfiguration von Vite, gemeinsam für alle Beispiele
// -------------------------------------------------------------------------
// Vite ist das Build-Tool von React: ein Entwicklungsserver
// (npm start) und ein Produktions-Compiler (npm run build). Jedes
// Beispiel ist ein Ordner mit eigener Seite index.html; sein
// Name wird an Vite übergeben, das ihn zum ROOT der Anwendung macht:
//
//     npm start 05-controlled-fields      ->  vite --config vite.config.ts 05-controlled-fields
//     npm run build 05-controlled-fields  ->  das Ergebnis in 05-controlled-fields/dist/
// =========================================================================

import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

export default defineConfig({
  // das React-Plugin: die JSX-Syntax und das „Hot“-Reloading (eine
  // Codeänderung aktualisiert die Seite, ohne den angezeigten Zustand zu verlieren)
  plugins: [react()],
  server: {
    port: 5173,
    // der PROXY: jede Anfrage des Browsers an /api/... wird an den
    // JSON-Server von RdvMedecins (Port 8080) weitergeleitet. Er wird erst ab
    // Beispiel 22 verwendet (vgl. Kapitel „Mit dem Server sprechen“).
    proxy: {
      '/api': 'http://localhost:8080',
    },
  },
});
    • Zeilen 9–10 (Kommentar): Der Name des Beispiels, der dem Befehl übergeben wird, wird zum Wurzelverzeichnis ([root]) der Anwendung: [Vite] sucht dort die Seite [index.html] und den Ordner [public/] und legt dort das Ergebnis des Builds ([dist/]) ab;
    • Zeile 19: das [React]-Plugin: Es kompiliert die JSX-Syntax (siehe unten) und aktiviert das Hot Reloading: Eine Änderung am Code einer Komponente wird auf die Seite angewendet, ohne sie neu zu laden, wobei der angezeigte Zustand so weit wie möglich erhalten bleibt;
    • Zeile 21: der Port des Entwicklungsservers;
    • Zeilen 25–27: der Proxy, der erst ab Beispiel 22 verwendet wird: Jede Anfrage des Browsers an eine Adresse, die mit [/api] beginnt, wird an den JSON-Server [RdvMedecins] weitergeleitet. Wir kommen später darauf zurück.

    // =========================================================================
// 01-first-application/src/App.tsx - eine erste Komponente
// -------------------------------------------------------------------------
// Eine React-Komponente ist eine FUNKTION, deren Name mit einem Großbuchstaben
// beginnt und die das, was angezeigt werden soll, ZURÜCKGIBT, geschrieben in JSX: HTML
// innerhalb des TypeScript-Codes. React ruft diese Funktion auf, um die
// Seite aufzubauen; es ruft sie bei jeder Änderung des Zustands der Komponente ERNEUT auf
// und aktualisiert die Seite mit dem neuen Ergebnis.
// =========================================================================

import { useState } from 'react';
// ein Stylesheet, importiert von der Komponente, die es verwendet
import './App.css';

// „gewöhnliche“ Daten außerhalb der Komponente: Sie ändern sich nie
const title = 'Welcome to React';
const languages = ['TypeScript', 'HTML', 'CSS'];

export default function App() {
  // useState(0): ein ZUSTAND mit dem Anfangswert 0. Wir erhalten seinen
  // AKTUELLEN Wert (counter) und die Funktion, die ihn ändert (setCounter).
  // Der Aufruf von setCounter veranlasst React, App ERNEUT aufzurufen, um
  // die Seite zu aktualisieren. Wir ändern die Seite nie selbst.
  const [counter, setCounter] = useState(0);
  const [time, setTime] = useState(() => new Date().toLocaleTimeString());

  // eine Funktion, die von der Schaltfläche aufgerufen wird
  function increment() {
    setCounter((n) => n + 1);
    setTime(new Date().toLocaleTimeString());
  }

  // das zurückgegebene JSX: { Ausdruck } fügt den Wert eines Ausdrucks ein;
  // className ersetzt das HTML-Attribut class (ein reserviertes Wort in JavaScript)
  return (
    <div className="container py-4">
      <h1 className="h3">{title}</h1>
      <p>This page is built in the browser, by React, from index.html, main.tsx and the App component.</p>
      <p>
        It is written in {languages.join(', ')} ({languages.length} languages).
      </p>

      {/* onClick: die Funktion, die beim Klicken aufgerufen wird (vgl. Beispiel 04) */}
      <button className="btn btn-primary" onClick={increment}>
        Click me
      </button>
      <p className="mt-3 counter">
        You clicked {counter} times (last click: {time}).
      </p>
    </div>
  );
}
    • Zeile 11: der Hook [useState], bereitgestellt von [React];
    • Zeile 13: das Stylesheet der Komponente;
    • Zeilen 16–17: zwei gewöhnliche Daten, die außerhalb der Komponente deklariert sind: Sie ändern sich nie;
    • Zeile 19: die Komponente: eine Funktion, deren Name mit einem Großbuchstaben beginnt (so unterscheidet JSX eine Komponente, [<App />], von einem HTML-Element, [<div>]). [export default]: Das exportiert das Modul standardmäßig; es wird in [main.tsx] mit [import App from './App'] importiert;
    • Zeile 24: [useState(0)] erzeugt eine Zustandsvariable mit dem Anfangswert 0. Es gibt ein Array mit zwei Elementen zurück, das wir „entpacken“: den aktuellen Wert ([counter]) und die Funktion, die ihn ändert ([setCounter]). Der Zustand wird von [React] gespeichert: Er bleibt von einem Funktionsaufruf zum nächsten erhalten;
    • Zeile 25: Sie können eine Funktion an [useState] übergeben, die den Anfangswert berechnet; sie wird nur beim ersten Anzeigen der Seite aufgerufen;
    • Zeilen 28–30: Eine Funktion, die vom Button aufgerufen wird. Sie ändert nicht die Seite, sondern den Zustand. [setCounter((n) => n + 1)] berechnet den neuen Wert anhand des alten. Jeder Aufruf der Funktion [setXxx] weist [React] an, [App] aufzurufen; [React] vergleicht das neue Ergebnis mit dem alten und aktualisiert nur die Teile der Seite, die sich geändert haben;
    • Zeilen 35–51: das zurückgegebene JSX, also das, was angezeigt werden soll:
      • JSX ähnelt HTML, mit einigen Unterschieden: Das Attribut [class] wird als [className] geschrieben ([class] ist ein reserviertes Wort in [JavaScript]), jedes Tag muss geschlossen werden ([<br />], [<input />]), und die gesamte Struktur muss EIN Wurzelelement haben;
      • Zeile 37: [{ expression }] fügt den Wert eines [TypeScript]-Ausdrucks ein. Er wird maskiert: Wenn [title] den Text [<b>] enthielte, würde [<b>] angezeigt, ohne dass das Tag interpretiert würde;
      • Zeile 40: Der Ausdruck kann beliebig sein: ein Methodenaufruf, eine Berechnung;
      • Zeile 43: Ein Kommentar in JSX wird als [{/* ... */}] geschrieben;
      • Zeile 44: Ein Klick auf den Button ruft die Funktion [increment] auf (siehe Beispiel 04). Die Funktion selbst wird ohne Klammern übergeben: [onClick={increment()}] würde sie sofort aufrufen, noch während der Anzeige;
      • Zeile 48: der aktuelle Wert des Zustands. Ändert sich der Zustand, wird [App] erneut aufgerufen, und dieser Text ändert sich.

    JSX ist nicht dasselbe wie HTML: Es ist eine bequeme Schreibweise für Funktionsaufrufe. Der Compiler wandelt [<h1 className="h3">{title}</h1>] in einen Aufruf der Form [jsx('h1', { className: 'h3', children: title })] um, der ein Objekt erzeugt, das das Element beschreibt. Die Komponente gibt also einen Baum von Objekten zurück (das „virtuelle DOM“), den [React] mit dem vorherigen Baum vergleicht, um zu bestimmen, was auf der tatsächlichen Seite geändert werden muss.

// 02-state/src/App.tsx - Zustand, berechnete Werte, useEffect
// -------------------------------------------------------------------------
// Ein kleiner Preisrechner einschließlich Mehrwertsteuer:
//   - useState    : ein ZUSTANDSwert; seine Änderung (setXxx) ruft die

// ein Formatierer für Eurobeträge nach englischer Schreibweise (€1,234.50)
const euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

export default function App() {
  const [label, setLabel] = useState('Pen');
  const [priceExclVat, setPriceExclVat] = useState(2.5);
  const [quantity, setQuantity] = useState(1);
  const [vatRate, setVatRate] = useState(20); // in %

  // BERECHNETE Werte: einfache Konstanten, die bei jedem
  // Aufruf der Komponente neu berechnet werden. Sie sind immer aktuell, nichts muss synchronisiert werden
  const totalExclVat = quantity * priceExclVat;
  const vatAmount = (totalExclVat * vatRate) / 100;
  // useMemo(Berechnung, [Abhängigkeiten]): dasselbe Ergebnis, aber React wiederholt die
  // Berechnung nur, wenn sich seit dem vorherigen Aufruf eine der Abhängigkeiten geändert hat
  const totalInclVat = useMemo(() => totalExclVat + vatAmount, [totalExclVat, vatAmount]);

  // der Verlauf der Summen inkl. MwSt.: Er hängt von den AUFEINANDERFOLGENDEN Werten der
  // Summe ab. Die alte Summe wird im Zustand gehalten; weicht sie von der
  // neuen ab, wird der Zustand WÄHREND des Aufrufs der Komponente angepasst (React
  // startet den Aufruf dann sofort neu, bevor irgendetwas angezeigt wird)
  const [history, setHistory] = useState<{ id: number; text: string }[]>([]);
  const [oldTotal, setOldTotal] = useState(totalInclVat);
  if (totalInclVat !== oldTotal) {
    setOldTotal(totalInclVat);
    const text = `${euros.format(oldTotal)} -> ${euros.format(totalInclVat)}`;
    // jede Zeile hat eine eindeutige Nummer (id), die sie identifiziert (vgl. key, Beispiel 03)
    setHistory((h) => [{ id: (h[0]?.id ?? 0) + 1, text }, ...h].slice(0, 5));
  }

  // useEffect(Effekt, [Abhängigkeiten]): der Effekt wird NACH der ersten
  // Anzeige ausgeführt, dann nach jeder Anzeige, bei der sich totalInclVat geändert hat
  useEffect(() => {
    document.title = `Incl. VAT: ${euros.format(totalInclVat)}`;
  }, [totalInclVat]);

  // Das Ändern des Artikels setzt die Menge auf 1 zurück: Das geschieht im
  // Ereignishandler, wo wir wissen, WARUM sich der Zustand ändert
  function changeLabel(newValue: string) {
    setLabel(newValue);
    setQuantity(1);
  }

  function reset() {
    setLabel('Pen');
    setPriceExclVat(2.5);
    setQuantity(1);
    setVatRate(20);
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 02 - state and computed values</h1>
      <div className="row g-4">
        <div className="col-md-6">
          <div className="card p-3">
            {/* value + onChange: das Feld zeigt den Zustand an, und jeder Tastenanschlag ändert ihn (vgl. Beispiel 05) */}
            <label className="form-label" htmlFor="label">Item</label>
            <input id="label" className="form-control mb-2" value={label}
                   onChange={(e) => changeLabel(e.target.value)} />
            <label className="form-label" htmlFor="price">Unit price excl. VAT</label>
            {/* valueAsNumber: der Wert des Feldes als Zahl (NaN, wenn es leer ist) */}
            <input id="price" type="number" step="0.1" min="0" className="form-control mb-2" value={priceExclVat}
                   onChange={(e) => setPriceExclVat(e.target.valueAsNumber || 0)} />
            <label className="form-label" htmlFor="quantity">Quantity (back to 1 if the item changes)</label>
            <input id="quantity" type="number" min="0" className="form-control mb-2" value={quantity}
                   onChange={(e) => setQuantity(e.target.valueAsNumber || 0)} />
            <label className="form-label" htmlFor="vat">VAT rate</label>
            {/* der Wert eines <select> ist ein STRING: Number(...) wandelt ihn um */}
            <select id="vat" className="form-select mb-3" value={vatRate}
                    onChange={(e) => setVatRate(Number(e.target.value))}>
              <option value={5.5}>5.5 %</option>
              <option value={10}>10 %</option>
              <option value={20}>20 %</option>
            </select>
            <button className="btn btn-outline-secondary" onClick={reset}>Reset</button>
          </div>
        </div>
        <div className="col-md-6">
          <div className="card p-3">
            <h2 className="h5">{quantity} x {label}</h2>
            <table className="table mb-0">
              <tbody>
                <tr><td>Total excl. VAT</td><td className="text-end">{euros.format(totalExclVat)}</td></tr>
                <tr><td>VAT ({vatRate} %)</td><td className="text-end">{euros.format(vatAmount)}</td></tr>
                <tr className="fw-bold"><td>Total incl. VAT</td><td className="text-end">{euros.format(totalInclVat)}</td></tr>
              </tbody>
            </table>
          </div>
          <div className="card p-3 mt-3">
            <h2 className="h6">History of the total incl. VAT</h2>
            <ul className="mb-0">
              {history.map((line) => (
                <li key={line.id}>{line.text}</li>
              ))}
            </ul>
          </div>
        </div>
      </div>
    </div>
  );
}

// -------------------------------------------------------------------------
// JSX ist HTML, das INNERHALB des TypeScript-Codes geschrieben wird. Jedes Tag wird zu
// einem Funktionsaufruf, und alles zwischen geschweiften Klammern ist ein gewöhnlicher

//   - attribute={expression}    : der Wert eines Attributs (src, disabled...);
//   - className={...}           : die CSS-Klassen, ein String, den wir berechnen;
//   - style={{ prop: value }}   : ein OBJEKT mit CSS-Eigenschaften (camelCase);
//   - {condition && <p/>}       : ein Element, das nur existiert, wenn die
//                                 Bedingung wahr ist;
//   - {c ? <A/> : <B/>}         : das eine ODER das andere;
//   - {list.map((x) => <li key={x.id}/>)} : ein Element pro Element der
//                                 Liste; key gibt jedem die IDENTITÄT.
// Keine besondere Syntax für Bedingungen und Schleifen: Es ist gewöhnliches
// TypeScript. Alles vor dem return ist gewöhnlicher Code.

import { useState } from 'react';

// ein TypeScript-Interface beschreibt die Form der Objekte der Liste
interface Doctor {
  id: number;
  title: string;
  firstName: string;
  lastName: string;
  specialty: 'general practitioner' | 'pediatrician' | 'cardiologist';
  available: boolean;
}

  { id: 1, title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER', specialty: 'general practitioner', available: true },
  { id: 2, title: 'Mr', firstName: 'Paul', lastName: 'TATOU', specialty: 'pediatrician', available: false },

];

function icon(specialty: Doctor['specialty']) {
  switch (specialty) {
    case 'pediatrician':
      return <span title="children">&#x1F9F8;</span>;
    case 'cardiologist':
      return <span title="heart">&#x2764;&#xFE0F;</span>;
    default:
      return <span title="all patients">&#x1FA7A;</span>;
  }
}

import { useRef, useState, type FormEvent, type KeyboardEvent, type MouseEvent } from 'react';

interface Task {
  id: number;
  label: string;
  done: boolean;
}

// der Stil des anklickbaren Bereichs: ein Objekt, wie beim Attribut style
const areaStyle = { background: '#e7f5ff', height: 80, cursor: 'crosshair' };

export default function App() {
  const [tasks, setTasks] = useState<Task[]>([
    { id: 1, label: 'Install Node.js', done: true },
    { id: 2, label: 'Read the chapter on events', done: false },
  ]);
  const [log, setLog] = useState<string[]>([]);
  // eine Referenz auf das <input> der neuen Aufgabe (null, solange es nicht angezeigt wird)
  const field = useRef<HTMLInputElement>(null);
  // eine Referenz kann auch einen Wert halten, der NICHT angezeigt wird: Ihn zu
  // ändern ruft die Komponente nicht erneut auf
  const nextId = useRef(3);

  function trace(message: string) {
    setLog((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // onSubmit: Eingabetaste im Feld ODER Klick auf die Schaltfläche. preventDefault()
  // verhindert, dass der Browser das Formular sendet (und die Seite neu lädt)
  function add(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const input = field.current!;
    const label = input.value.trim();
    if (label === '') {
      return;
    }
    // ein Zustand, der ein Array enthält: Er wird durch ein neues Array ERSETZT
    // (das alte wird nie verändert: React würde die Änderung nicht bemerken)
    setTasks((t) => [...t, { id: nextId.current++, label, done: false }]);
    trace(`added "${label}"`);
    input.value = '';
    input.focus();
  }

  // eine Taste: event.key liefert ihren Namen
  function onKeyDown(event: KeyboardEvent<HTMLInputElement>) {
    if (event.key === 'Escape') {
      event.currentTarget.value = '';
    }
  }

  function toggle(task: Task, done: boolean) {
    setTasks((t) => t.map((x) => (x.id === task.id ? { ...x, done } : x)));
    trace(`"${task.label}": ${done ? 'done' : 'to do'}`);
  }

  function remove(task: Task) {
    setTasks((t) => t.filter((x) => x.id !== task.id));
    trace(`removed "${task.label}"`);
  }

  // das ursprüngliche Ereignis (nativeEvent) enthält die Koordinaten relativ zum Bereich
  function position(event: MouseEvent<HTMLDivElement>) {
    trace(`click at (${event.nativeEvent.offsetX}, ${event.nativeEvent.offsetY})`);

// 05-controlled-fields/src/App.tsx - alle Feldtypen, „kontrolliert“
// -------------------------------------------------------------------------
// Ein KONTROLLIERTES Feld zeigt einen Wert des Zustands an (value={...}) und meldet
// der Komponente jede Änderung (onChange={...}), die den
// Zustand aktualisiert. Der Zustand ist die einzige Quelle der Wahrheit: Eine Eingabe im Feld ändert
// den Zustand, und eine Änderung des Zustands (per Code) ändert das Feld.
// Es ist das „Two-Way“-Binding, von Hand geschrieben. Was im Ereignis gelesen
// wird, hängt vom Feld ab:
//   - text input, textarea           -> e.target.value (String)
//   - input type="number"            -> e.target.valueAsNumber (NaN, wenn leer)
//   - checkbox                       -> e.target.checked (boolean)
//   - radio buttons                  -> e.target.value der gewählten Schaltfläche
//   - select                         -> e.target.value
//   - select multiple                -> e.target.selectedOptions (mehrere)
//   - input type="date"              -> e.target.value, "yyyy-MM-dd"
// Dasselbe Formular wie Beispiel 18 des NestJS-Kurses; aber hier gibt es keinen
// Server: Alles geschieht im Browser.
// =========================================================================

import { useState, type FormEvent } from 'react';

const TITLES = ['Mr', 'Mme', 'Melle'];
const HOBBIES = ['sport', 'music', 'reading', 'cinema'];
const COUNTRIES = [
  { code: 'fr', label: 'France' },
  { code: 'be', label: 'Belgium' },
  { code: 'ch', label: 'Switzerland' },
];

export default function App() {
  // ein Zustand pro Feld des Formulars
  const [title, setTitle] = useState('Mme');
  const [name, setName] = useState('');
  const [age, setAge] = useState<number | null>(30);
  const [presentation, setPresentation] = useState('');
  const [newsletter, setNewsletter] = useState(false);
  const [hobbies, setHobbies] = useState<string[]>(['reading']);
  const [country, setCountry] = useState('fr');
  const [languages, setLanguages] = useState<string[]>([]);
  const [birthDate, setBirthDate] = useState('');
  const [sent, setSent] = useState<string | null>(null);

  // alle Werte, in einem Objekt zusammengefasst (was an einen Server gesendet würde)
  const model = { title, name: name.trim(), age, presentation, newsletter, hobbies, country, languages, birthDate };

  // mehrere Kästchen für EIN Array: Der Wert wird hinzugefügt oder entfernt
  function toggleHobby(hobby: string, checked: boolean) {
    setHobbies((l) => (checked ? [...l, hobby] : l.filter((x) => x !== hobby)));
  }

  function send(event: FormEvent<HTMLFormElement>) {
    event.preventDefault(); // kein Senden an den Server, kein Neuladen
    setSent(JSON.stringify(model));
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 05 - controlled fields</h1>
      <div className="row g-4">
        <form className="col-md-7" onSubmit={send}>
          <div className="mb-3">
            <span className="form-label d-block">Title</span>
            {/* Radiobuttons: derselbe Zustand; checked gibt an, welcher ausgewählt ist */}
            {TITLES.map((t) => (
              <div key={t} className="form-check form-check-inline">
                <input type="radio" className="form-check-input" name="title" id={`title-${t}`} value={t}
                       checked={title === t} onChange={(e) => setTitle(e.target.value)} />
                <label className="form-check-label" htmlFor={`title-${t}`}>{t}</label>
              </div>
            ))}
          </div>
          <div className="row g-3 mb-3">
            <div className="col-8">
              <label className="form-label" htmlFor="name">Name</label>
              <input id="name" className="form-control" value={name} onChange={(e) => setName(e.target.value)} />
            </div>
            <div className="col-4">
              <label className="form-label" htmlFor="age">Age</label>
              {/* ein leeres Feld liefert NaN: Im Zustand wird null gespeichert, im Feld '' */}
              <input id="age" type="number" className="form-control" value={age ?? ''}

            </div>
          </div>
          <div className="mb-3">

// 06-use-reducer/src/App.tsx - ein Formular, ein Objekt, ein Reducer
// -------------------------------------------------------------------------
// Beispiel 05 hatte ein useState pro Feld. Für ein echtes Formular
// bevorzugen wir EIN Objekt, das alle Werte enthält, das MODELL, und
// beschreiben seine Änderungen in einer einzigen Funktion, dem REDUCER:
//
//     reducer(currentState, action) -> newState
//
// useReducer(reducer, initialState) liefert den Zustand und eine Funktion
// dispatch(action): Die Komponente ändert den Zustand nicht mehr selbst, sie BESCHREIBT, was
// geschehen ist ({ type: 'edit', field: 'lastName', value: 'Dupont' }), und der
// Reducer leitet daraus den neuen Zustand ab. Die gesamte Logik steht an einer Stelle,
// und es ist eine REINE Funktion (gleiche Eingabe, gleiche Ausgabe, kein
// Seiteneffekt), leicht zu testen.
//
// Der Zustand merkt sich außerdem für jedes Feld, ob es „berührt“ wurde (mindestens
// einmal verlassen); daraus leiten wir ab, ob es „geändert“ ist (sein Wert weicht vom
// Anfangswert ab) und ob es deaktiviert ist.
// =========================================================================

import { useReducer, type ChangeEvent } from 'react';

// die Form der Daten des Formulars
interface Profile {
  title: string;
  firstName: string;
  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;
}

const INITIAL: Profile = {
  title: 'Mme',
  firstName: 'Marie',
  lastName: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  country: 'fr',
  birthDate: '',
  presentation: '',
};

type Field = keyof Profile; // 'title' | 'firstName' | ...: der Name eines Feldes

interface State {
  values: Profile;
  touched: Partial<Record<Field, boolean>>; // die bereits verlassenen Felder
}

// die möglichen Aktionen: eine „diskriminierte Union“ über die Eigenschaft type
type Action =
  | { type: 'edit'; field: Field; value: Profile[Field] }
  | { type: 'leave'; field: Field }
  | { type: 'uppercase' }
  | { type: 'reset' };

// der Reducer: Er verändert den erhaltenen Zustand NIE, er gibt einen neuen zurück
function reducer(state: State, action: Action): State {
  switch (action.type) {
    case 'edit':
      return { ...state, values: { ...state.values, [action.field]: action.value } };
    case 'leave':
      return { ...state, touched: { ...state.touched, [action.field]: true } };
    case 'uppercase': {
      const { firstName, lastName } = state.values;
      return { ...state, values: { ...state.values, firstName: firstName.toUpperCase(), lastName: lastName.toUpperCase() } };
    }
    case 'reset':
      return { values: { ...INITIAL }, touched: {} };
  }
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const TRACKED_FIELDS: Field[] = ['firstName', 'lastName', 'age', 'email'];

export default function App() {
  const [state, dispatch] = useReducer(reducer, { values: { ...INITIAL }, touched: {} });
  const v = state.values;

  // aus dem Zustand ABGELEITETE Werte
  const modified = (c: Field) => v[c] !== INITIAL[c];
  const disabled = (c: Field) => c === 'email' && !v.newsletter; // die E-Mail setzt den Newsletter voraus
  const formModified = (Object.keys(INITIAL) as Field[]).some(modified);

  // die für ein Textfeld gemeinsamen Attribute: eine Funktion, die sie aufbaut
  // (id, value, onChange, onBlur, disabled), um sie auf das Element zu „spreaden“: {...bind('lastName')}
  function bind(field: 'firstName' | 'lastName' | 'email' | 'birthDate' | 'presentation' | 'country') {
    return {
      id: field,
      value: v[field],
      disabled: disabled(field),
      onChange: (e: ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) =>
        dispatch({ type: 'edit', field, value: e.target.value }),
      onBlur: () => dispatch({ type: 'leave', field }),
    };
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 06 - a form, an object, a reducer</h1>
      <div className="row g-4">
        <form className="col-md-7" onSubmit={(e) => e.preventDefault()}>
          <div className="mb-3">
            <span className="form-label d-block">Title</span>
            {TITLES.map((t) => (
              <div key={t} className="form-check form-check-inline">
                <input type="radio" className="form-check-input" id={`title-${t}`} name="title" value={t}
                       checked={v.title === t} onChange={() => dispatch({ type: 'edit', field: 'title', value: t })} />
                <label className="form-check-label" htmlFor={`title-${t}`}>{t}</label>
              </div>
            ))}
          </div>
          <div className="row g-3 mb-3">
            <div className="col-5">
              <label className="form-label" htmlFor="firstName">First name</label>
              {/* {...object}: jede Eigenschaft des Objekts wird zu einem Attribut */}
              <input className="form-control" {...bind('firstName')} />
            </div>
            <div className="col-5">
              <label className="form-label" htmlFor="lastName">Last name</label>
              <input className="form-control" {...bind('lastName')} />
            </div>
            <div className="col-2">
              <label className="form-label" htmlFor="age">Age</label>
              <input id="age" type="number" className="form-control" value={v.age}
                     onChange={(e) => dispatch({ type: 'edit', field: 'age', value: e.target.valueAsNumber || 0 })}
                     onBlur={() => dispatch({ type: 'leave', field: 'age' })} />
            </div>
          </div>
          <div className="row g-3 mb-3 align-items-end">
            <div className="col-5">
              <div className="form-check">
                <input id="newsletter" type="checkbox" className="form-check-input" checked={v.newsletter}
                       onChange={(e) => dispatch({ type: 'edit', field: 'newsletter', value: e.target.checked })} />
                <label htmlFor="newsletter" className="form-check-label">Receive the newsletter</label>
              </div>
            </div>
            <div className="col-7">
              <label className="form-label" htmlFor="email">E-mail (if newsletter)</label>
              <input type="email" className="form-control" {...bind('email')} />
            </div>
          </div>
          <div className="row g-3 mb-3">

// 07-validation/src/App.tsx - ein Formular im Browser validieren
// -------------------------------------------------------------------------
// React hat kein Validierungssystem: Es ist gewöhnlicher Code. Eine
// REINE Funktion, validate(account), liefert die Fehler der Felder; sie wird
// bei jeder Anzeige aufgerufen, die Fehler sind also immer aktuell.
// Sie werden nicht sofort angezeigt (ein komplett rot markiertes leeres Formular
// ist unangenehm): nur für ein bereits verlassenes („berührtes“) Feld. Das Absenden
// markiert alle Felder als berührt und handelt dann nur, wenn das Formular
// gültig ist.
//
// Dieselben Regeln wie beim RdvMedecins-Server (Formular „Konto erstellen“).
// ACHTUNG: Diese Validierung ist nur eine ERLEICHTERUNG für den Benutzer (eine
// sofortige Antwort, ohne Hin und Zurück zum Server). Sie schützt
// nichts: Jeder kann dem Server eine von Hand gebaute Anfrage
// senden (curl...). Der Server validiert daher IMMER alles erneut (vgl. Beispiel 25).

import { useState, type FormEvent } from 'react';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

type Field = keyof Account;
// höchstens eine Fehlermeldung pro Feld
type Errors = Partial<Record<Field, string>>;

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

// die Regeln: eine reine Funktion, die nur von ihren Argumenten abhängt
function validate(c: Account): Errors {
  const errors: Errors = {};
  if (!TITLES.includes(c.title)) errors.title = 'Choose a title from the list.';
  if (c.firstName.trim() === '') errors.firstName = 'The first name is required.';
  else if (c.firstName.length > 20) errors.firstName = 'The first name must not exceed 20 characters.';
  if (c.lastName.trim() === '') errors.lastName = 'The last name is required.';
  else if (c.lastName.length > 20) errors.lastName = 'The last name must not exceed 20 characters.';
  if (c.login === '') errors.login = 'The login is required.';
  else if (!LOGIN_FORMAT.test(c.login)) errors.login = 'Unaccented lower-case letters, digits, dot, hyphen, underscore.';
  if (c.password === '') errors.password = 'The password is required.';
  else if (c.password.length < PASSWORD_MIN) errors.password = `At least ${PASSWORD_MIN} characters.`;
  // eine Regel, die ZWEI Felder vergleicht
  if (c.confirmation !== c.password) errors.confirmation = 'The confirmation does not match the password.';
  return errors;
}

const EMPTY: Account = { title: 'Mr', firstName: '', lastName: '', login: '', password: '', confirmation: '' };
const ALL: Record<Field, boolean> = { title: true, firstName: true, lastName: true, login: true, password: true, confirmation: true };

export default function App() {
  const [account, setAccount] = useState<Account>(EMPTY);
  const [touched, setTouched] = useState<Partial<Record<Field, boolean>>>({});
  const [submitAttempted, setSubmitAttempted] = useState(false);
  const [success, setSuccess] = useState(false);

  // die Fehler: bei jeder Anzeige neu berechnet, nie im Zustand gespeichert
  const errors = validate(account);
  const errorCount = Object.keys(errors).length;

  // die Bootstrap-Klasse eines Feldes: rot (is-invalid) oder grün (is-valid),
  // erst nachdem es berührt wurde
  function cssClass(field: Field, base = 'form-control'): string {
    if (!touched[field]) return base;
    return `${base} ${errors[field] ? 'is-invalid' : 'is-valid'}`;
  }

  // die allen Feldern gemeinsamen Attribute
  function bind(field: Field) {
    return {
      id: field,
      value: account[field],
      onChange: (e: { target: { value: string } }) => {
        setAccount((c) => ({ ...c, [field]: e.target.value }));
        setSuccess(false);
      },
      onBlur: () => setTouched((t) => ({ ...t, [field]: true })),
    };
  }

  function send(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setSubmitAttempted(true);
    setTouched(ALL); // alle Fehler erscheinen
    // hier, das Konto würde an den Server gesendet (vgl. Beispiel 25)
    setSuccess(errorCount === 0);
  }

  return (
    <div className="container py-4" style={{ maxWidth: 640 }}>
      <h1 className="h3 mb-3">Example 07 - the validation of a form</h1>
      {/* noValidate: die eigene Validierung des Browsers ist deaktiviert, um
          UNSERE Meldungen in UNSEREM Stil anzuzeigen */}
      <form className="card p-4" noValidate onSubmit={send}>
        <div className="mb-3">
          <label className="form-label" htmlFor="title">Title</label>
          <select className={cssClass('title', 'form-select')} {...bind('title')}>
            {TITLES.map((t) => <option key={t}>{t}</option>)}
          </select>
          <div className="invalid-feedback">{errors.title}</div>
        </div>
        <div className="row g-3 mb-3">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="firstName">First name</label>
            {/* is-invalid umrandet das Feld rot UND zeigt die
                darauffolgende .invalid-feedback an: dort steht der Fehler des Felds */}
            <input className={cssClass('firstName')} {...bind('firstName')} />
            <div className="invalid-feedback">{errors.firstName}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="lastName">Last name</label>
            <input className={cssClass('lastName')} {...bind('lastName')} />
            <div className="invalid-feedback">{errors.lastName}</div>
          </div>
        </div>
        <div className="mb-3">
          <label className="form-label" htmlFor="login">Login</label>
          <input className={cssClass('login')} autoComplete="username" {...bind('login')} />
          <div className="invalid-feedback">{errors.login}</div>
        </div>
        <div className="row g-3 mb-4">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="password">Password</label>
            <input type="password" className={cssClass('password')} autoComplete="new-password" {...bind('password')} />
            <div className="invalid-feedback">{errors.password}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="confirmation">Confirmation</label>
            <input type="password" className={cssClass('confirmation')} autoComplete="new-password" {...bind('confirmation')} />
            <div className="invalid-feedback">{errors.confirmation}</div>
          </div>
        </div>
        <button type="submit" className="btn btn-primary">Create the account</button>
        {submitAttempted && errorCount > 0 && (
          <div className="alert alert-danger mt-3 mb-0">The form contains {errorCount} error(s).</div>
        )}
        {success && <div className="alert alert-success mt-3 mb-0">Valid form: it could be sent to the server.</div>}
      </form>
    </div>
  );
}

import { useForm } from 'react-hook-form';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITLES = ['Mr', 'Mme', 'Melle'];

export default function App() {

    register,
    handleSubmit,
    getValues,
    formState: { errors, isSubmitted },
  } = useForm<Account>({
    // die Anfangswerte

  // es erhält die Werte, typisiert

    setSent(JSON.stringify(account));
  }

  return (

                     required: 'The last name is required.',

                   })} />

          </div>

// 09-formatting/src/formatting.ts - „selbstgebaute“ Formatierungsfunktionen
// -------------------------------------------------------------------------
// React hat keinen eigenen Mechanismus zur Formatierung (Angular hat seine
// „Pipes“, Vue.js seine Funktionen): In JSX rufen wir gewöhnliche
// FUNKTIONEN auf, die in einem Modul liegen und für sich testbar sind:
//     {timeRange(slot)}           -> "8:00 - 8:20"
//     {timeRange(slot, 'start')}  -> "8:00"
// Für Datumsangaben, Zahlen und Beträge stellt der Browser die
// Intl-API bereit, die die Konventionen jeder Sprache kennt.
// =========================================================================

export interface Slot {
  startHour: number;
  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" statt "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

export function timeRange(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
  switch (part) {
    case 'start':
      return time(c.startHour, c.startMinute);
    case 'end':
      return time(c.endHour, c.endMinute);
    default:
      return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
  }
}

// "marie pelissier" -> "Marie Pelissier"
export function capitalize(text: string): string {
  return text.replace(/\p{L}+/gu, (word) => word[0].toUpperCase() + word.slice(1).toLowerCase());
}

// "Marie PELISSIER" -> "MP"
export function initials(fullName: string): string {
  return fullName
    .split(/\s+/)
    .filter((word) => word !== '')
    .map((word) => word[0].toUpperCase())
    .join('');
}

// "2026-10-05" (das Format der RdvMedecins-API) -> ein Date, um Mitternacht UTC
export function toDate(day: string): Date {
  return new Date(`${day}T00:00:00Z`);
}
    • Zeilen 13–18: ein Zeitfenster, wie es vom [RdvMedecins]-Server gesendet wird (Stunden und Minuten von Beginn und Ende);
    • Zeilen 21–23: [padStart(2, '0')] füllt die Zeichenkette links auf: „5“ wird zu „05“;
    • Zeilen 25–34: „8:00 – 8:20“ oder nur der Beginn oder das Ende, je nach Parameter [part], der optional ist (Standardwert ['all']);
    • Zeilen 37–39: [\p{L}+] bezeichnet eine Folge von Buchstaben, einschließlich Buchstaben mit Akzent (Option [u]): Der erste Buchstabe wird großgeschrieben;
    • Zeilen 51–53: ein Datum in der Form der API (["2026-10-05"]) wird zu einem [Date]-Objekt um Mitternacht UTC.

// 09-formatting/src/App.tsx - Werte in JSX formatieren
// -------------------------------------------------------------------------
// Jede Zeile der Tabelle zeigt ein Stück Code und das, was es anzeigt.
// Die Intl-Formatierer werden einmalig außerhalb der Komponente erstellt:
// Sie zu erstellen kostet mehr, als sie zu verwenden.
// =========================================================================

import { useState } from 'react';
import { timeRange, initials, capitalize, toDate, type Slot } from './formatting';

// Datumsangaben: dateStyle / timeStyle oder die Angabe der anzuzeigenden Teile
const shortDate = new Intl.DateTimeFormat('en-US', { dateStyle: 'medium' });
const fullDate = new Intl.DateTimeFormat('en-US', { dateStyle: 'full' });
const timeOnly = new Intl.DateTimeFormat('en-US', { timeStyle: 'medium' });
const longDay = new Intl.DateTimeFormat('en-US', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric', timeZone: 'UTC' });
const frenchDay = new Intl.DateTimeFormat('fr-FR', { dateStyle: 'long', timeZone: 'UTC' });
// Zahlen und Beträge
const euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });
const integer = new Intl.NumberFormat('en-US', { maximumFractionDigits: 0 });
const percent = new Intl.NumberFormat('en-US', { style: 'percent', minimumFractionDigits: 1 });

const day = '2026-10-05'; // ein Datum im Format der RdvMedecins-API
const price = 1234.5;
const rate = 0.055;
const name = 'marie pelissier';
const doctor = { title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER' };
const languages = ['TypeScript', 'HTML', 'CSS', 'JavaScript', 'SQL'];
const slots: Slot[] = [
  { startHour: 8, startMinute: 0, endHour: 8, endMinute: 20 },
  { startHour: 8, startMinute: 20, endHour: 8, endMinute: 40 },
  { startHour: 14, startMinute: 5, endHour: 14, endMinute: 25 },
];

export default function App() {
  const [now, setNow] = useState(() => new Date());

  // [angezeigter Code, Wert]: eine Zeile der Tabelle
  const rows: [string, string][] = [
    ['shortDate.format(now)', shortDate.format(now)],
    ['fullDate.format(now)', fullDate.format(now)],
    ['timeOnly.format(now)', timeOnly.format(now)],
    ['longDay.format(toDate(day))', longDay.format(toDate(day))],
    ['frenchDay.format(toDate(day))', frenchDay.format(toDate(day))],
    ['now.toLocaleDateString()', now.toLocaleDateString()],
    ['euros.format(price)', euros.format(price)],
    ['integer.format(price)', integer.format(price)],
    ['percent.format(rate)', percent.format(rate)],
    ['name.toUpperCase()', name.toUpperCase()],
    ['capitalize(name)', capitalize(name)],
    ["'RDV'.toLowerCase()", 'RDV'.toLowerCase()],
    ["languages.slice(0, 3).join(', ')", languages.slice(0, 3).join(', ')],
    ['JSON.stringify(doctor)', JSON.stringify(doctor)],
    // zwei Funktionen hintereinander: Das Ergebnis der ersten geht an die zweite
    ['initials(capitalize(name))', initials(capitalize(name))],
  ];

  return (
    <div className="container py-4" style={{ maxWidth: 820 }}>
      <h1 className="h3 mb-3">Example 09 - the formatting of values</h1>
      <table className="table table-sm align-middle">
        <thead><tr><th>In the code</th><th>Displayed</th></tr></thead>
        <tbody>
          {rows.map(([code, value]) => (
            <tr key={code}><td><code>{`{${code}}`}</code></td><td>{value}</td></tr>
          ))}
        </tbody>
      </table>

      <h2 className="h6">Object.entries: going through the properties of an object</h2>
      <ul>
        {Object.entries(doctor).map(([key, value]) => (
          <li key={key}>{key} = {value}</li>
        ))}
      </ul>

      <h2 className="h6">A home-made function: timeRange</h2>
      <ul>
        {slots.map((c) => (
          <li key={timeRange(c)}>{timeRange(c)} (start: {timeRange(c, 'start')})</li>
        ))}
      </ul>
      <button className="btn btn-sm btn-outline-primary" onClick={() => setNow(new Date())}>Refresh the time</button>
    </div>
  );
}
    • Zeilen 13–17: Datumsformatierer, [Intl.DateTimeFormat(language, options)]: [dateStyle] und [timeStyle] liefern vordefinierte Formate ([full], [long], [medium], [short]); man kann auch angeben, welche Teile angezeigt werden sollen ([weekday], [day], [month], [year]). [timeZone: 'UTC']: Das Datum [2026-10-05T00:00:00Z] muss unabhängig von der Zeitzone des Browsers der 5. Oktober bleiben;
    • Zeilen 19–21: Zahlenformatierer: ein Betrag in Euro, eine Zahl ohne Nachkommastellen, ein Prozentsatz;
    • Zeilen 39–56: Tabellenzeilen: der Code und sein Wert. Die letzte verkettet zwei Funktionen;
    • Zeile 65: der Code in geschweiften Klammern;
    • Zeilen 72–74: [Object.entries(object)] liefert die Paare [key, value] eines Objekts, die mit [map] durchlaufen werden;
    • Zeilen 79–81: die eigene Funktion [timeRange].

    Die [Intl]-Generatoren werden ein für alle Mal außerhalb der Komponente erzeugt: Ihre Erzeugung kostet mehr als ihre Verwendung. Die Sprache ist dort fest vorgegeben ([en-US], und [fr-FR] für die französische Zeile); Beispiel 21 zeigt, wie man der vom Benutzer gewählten Sprache folgt.
