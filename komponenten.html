    // =========================================================================
// 10-components/src/DoctorCard.tsx - eine KIND-Komponente
// -------------------------------------------------------------------------
// Sie erhält ihre Daten von ihrem Elternteil über PROPS (Eigenschaften): die
// Attribute, die der Elternteil auf <DoctorCard ... /> schreibt und die React
// in EINEM Objekt bündelt, dem ersten Parameter der Funktion. Ihr Typ wird
// zur Kompilierzeit geprüft: <DoctorCard doctor={42} /> würde abgelehnt.
//
// Goldene Regel: Eine Komponente ändert ihre Props NIEMALS. Die Daten
// fließen vom Elternteil zum Kind; um eine Änderung anzufordern,
// ruft das Kind eine Funktion auf, die der Elternteil ihm übergeben hat (vgl. Beispiel 11).
// =========================================================================

import type { Doctor } from './doctor';
// styles: ein Objekt { initials: '_initials_x7f3a_1' } - der tatsächliche Name jeder Klasse
import styles from './DoctorCard.module.css';

// der Typ der Props: was der Elternteil liefern KANN (?) oder MUSS
interface Props {
  doctor: Doctor;
  rank: number;
  compact?: boolean; // optional
}

// das Props-Objekt wird „destrukturiert“; compact = false: sein Standardwert
export default function DoctorCard({ doctor, rank, compact = false }: Props) {
  // ein aus den Props berechneter Wert: wird neu berechnet, wenn der Elternteil sie ändert
  const initials = `${doctor.firstName[0]}${doctor.lastName[0]}`;

  return (
    <div className={`card h-100 ${doctor.available ? 'border-success' : ''}`}>
      <div className="card-body d-flex gap-3 align-items-center">
        <div className={styles.initials}>{initials}</div>
        <div>
          <h2 className="h6 mb-1">{rank}. {doctor.title} {doctor.firstName} {doctor.lastName}</h2>
          {!compact && <p className="mb-0 text-muted small">{doctor.specialty}</p>}
        </div>
      </div>
    </div>
  );
}
    • Zeile 14: ein reiner Typ-Import;
    • Zeile 16: ein Import des CSS-Moduls (siehe unten);
    • Zeilen 19–23: die Typen der Props der Komponente – der Attribute, die die Elternkomponente in [<DoctorCard ... />] schreibt. [compact?] ist optional;
    • Zeile 26: [React] fasst die Props zu EINEM Objekt zusammen, das es der Funktion als ersten Parameter übergibt. Dieses wird sofort „entpackt“: [{ doctor, rank, compact = false }] ergibt drei Variablen, wobei [compact] den Wert [false] hat, wenn die Elternkomponente es nicht angibt. Der Typ der Props wird zur Kompilierzeit geprüft: [<DoctorCard doctor={42} />] würde abgelehnt, ebenso das Weglassen von [rank];
    • Zeile 28: ein aus den Props berechneter Wert: Ändert die Elternkomponente die Props, wird die Komponente erneut aufgerufen und der Wert neu berechnet;
    • Zeile 33: die CSS-Klasse, gelesen aus dem CSS-Modul;
    • Zeile 36: Die Fachrichtung wird nur im normalen Modus angezeigt.

    Goldene Regel: Eine Komponente ändert niemals ihre eigenen Props. Die Daten fließen von der Eltern- zur Kindkomponente; um eine Änderung anzufordern, ruft die Kindkomponente eine Funktion auf, die ihr die Elternkomponente übergeben hat (siehe Beispiel 11).

    // =========================================================================
// 10-components/src/App.tsx - die ELTERN-Komponente
// -------------------------------------------------------------------------
// Sie importiert die Komponente DoctorCard und verwendet sie als neues
// Tag, <DoctorCard />, so oft wie nötig: Eine Komponente ist
// ein WIEDERVERWENDBARES Stück Oberfläche mit eigenen Daten, eigenem JSX und
// eigenem Stil. Ein Tag, das mit einem GROSSBUCHSTABEN beginnt, ist eine Komponente;
// in Kleinbuchstaben (<div>, <p>) ist es ein HTML-Element.
// =========================================================================

import { useState } from 'react';
import DoctorCard from './DoctorCard';
import type { Doctor } from './doctor';

export default function App() {
  const [doctors, setDoctors] = useState<Doctor[]>([
    { id: 1, title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER', specialty: 'general practitioner', available: true },
    { id: 2, title: 'Mr', firstName: 'Paul', lastName: 'TATOU', specialty: 'pediatrician', available: false },
    { id: 3, title: 'Melle', firstName: 'Sophie', lastName: 'DUPONT', specialty: 'general practitioner', available: true },
  ]);
  const [compact, setCompact] = useState(false);

  // der Elternteil ändert SEINE Daten: React ruft den Elternteil erneut auf, dann die
  // Kinder, mit ihren neuen Props
  function toggleAvailability() {
    setDoctors((list) => list.map((m) => ({ ...m, available: !m.available })));
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 10 - components and their props</h1>
      <div className="d-flex gap-3 mb-3">
        <label className="form-check-label">
          <input type="checkbox" className="form-check-input" checked={compact}
                 onChange={(e) => setCompact(e.target.checked)} /> compact display
        </label>
        <button className="btn btn-sm btn-outline-primary" onClick={toggleAvailability}>Toggle availabilities</button>
      </div>
      <div className="row g-3">
        {doctors.map((m, i) => (
          <div key={m.id} className="col-md-4">
            {/* doctor={m} übergibt das OBJEKT m (einen Ausdruck);
                doctor="m" ohne geschweifte Klammern würde die ZEICHENKETTE "m" übergeben */}
            <DoctorCard doctor={m} rank={i + 1} compact={compact} />
          </div>
        ))}
      </div>
      <p className="mt-3 mb-1 small text-muted">The same card, in compact version, by an attribute without a value:</p>
      <div className="w-50">
        {/* compact ohne Wert ist true */}
        <DoctorCard doctor={doctors[0]} rank={1} compact />
      </div>
    </div>
  );
}
    • Zeile 12: Die Kindkomponente wird wie jedes andere Modul importiert;
    • Zeilen 25–27: Die Elternkomponente ändert IHRE Daten. [React] rendert sie neu und rendert dann die Kindkomponenten mit ihren neuen Props neu: Die Karten werden aktualisiert;
    • Zeile 44: Die Komponente wird als neues Tag verwendet. [doctor={m}] übergibt das OBJEKT [m] (ein Ausdruck); [doctor="m"] würde ohne geschweifte Klammern die ZEICHENKETTE „m“ übergeben;
    • Zeile 51: Ein Attribut ohne Wert ([compact]) entspricht [true].

    // =========================================================================
// 11-lifting-state-up/src/Counter.tsx
// -------------------------------------------------------------------------
// Ein Kind, das seinen Elternteil BENACHRICHTIGT: Der Elternteil übergibt ihm FUNKTIONEN
// in seinen Props (ihr Name beginnt per Konvention mit „on“), und das Kind
// ruft sie auf, wenn etwas geschieht:
//     <Counter label="Coffees" max={3} onIncrement={...} onLimit={...} />
// =========================================================================

import { useState } from 'react';

interface Props {
  label: string;
  max: number;
  onIncrement: (value: number) => void; // mit dem neuen Wert
  onLimit: () => void; // ohne Daten
}

export default function Counter({ label, max, onIncrement, onLimit }: Props) {
  // der EIGENE Zustand dieses Zählers: Jeder <Counter> hat seinen eigenen
  const [value, setValue] = useState(0);

  function plus() {
    if (value >= max) {
      onLimit();
      return;
    }
    setValue(value + 1);
    // value ist noch nicht geändert (das geschieht beim nächsten Aufruf der
    // Komponente): value + 1 wird übergeben
    onIncrement(value + 1);
  }

  return (
    <div className="d-flex align-items-center gap-2 mb-2">
      <span className="w-25">{label}</span>
      <button className="btn btn-sm btn-outline-primary" onClick={plus}>+1</button>
      <span className="badge text-bg-secondary">{value} / {max}</span>
    </div>
  );
}
    • Zeilen 12–17: Zwei Props sind Funktionen: [onIncrement] erhält den neuen Wert, während [onLimit] nichts erhält. Ihre Namen beginnen per Konvention mit „on“, wie bei [onClick];
    • Zeile 21: der EIGENE Zustand dieses Zählers: Jeder [<Counter>] auf der Seite hat seinen eigenen. Der Zustand ist an die Position der Komponente im Seitenbaum gebunden;
    • Zeilen 23–32: Die Elternkomponente wird benachrichtigt. Achtung: [setValue(value + 1)] ändert die Variable [value] nicht (eine Konstante, die für DIESEN Aufruf der Komponente festgelegt ist): Sie lautet beim NÄCHSTEN Aufruf [value + 1]. Daher wird [value + 1] an die Elternkomponente übergeben.

    // =========================================================================
// 11-lifting-state-up/src/TextField.tsx - eine KONTROLLIERTE Komponente
// -------------------------------------------------------------------------
// Ein „selbstgebautes“ Formularfeld (Label + Feld + Hilfe +
// Zeichenzähler), das wie ein kontrolliertes <input> verwendet wird (vgl. Beispiel 05):
//     <TextField value={lastName} onChange={setLastName} label="Last name" max={20} />
// Die Komponente hat KEINEN Zustand: Der Wert gehört dem Elternteil, der
// ihn übergibt (value) und den sie über jeden Tastenanschlag benachrichtigt (onChange). Es ist
// der Elternteil, der über den neuen Wert entscheidet.
// =========================================================================

import { useId } from 'react';

interface Props {
  value: string;
  onChange: (value: string) => void;
  label: string;
  max: number;
  help?: string;
}

export default function TextField({ value, onChange, label, max, help = '' }: Props) {
  // useId: ein EINDEUTIGER Bezeichner, verschieden für jedes <TextField> der
  // Seite, um <label htmlFor> und <input id> zu verknüpfen
  const id = useId();

  return (
    <div className="mb-3">
      <label className="form-label" htmlFor={id}>{label}</label>
      <input id={id} className={`form-control ${value.length === max ? 'is-invalid' : ''}`} value={value}
             maxLength={max} onChange={(e) => onChange(e.target.value)} />
      <div className="form-text">{help} ({value.length} / {max})</div>
    </div>
  );
}

    Ein „benutzerdefiniertes“ Formularfeld (Label, Feld, Tooltip, Zeichenzähler), das als kontrolliertes [<input>] verwendet wird (siehe Beispiel 05): Die Elternkomponente übergibt ihm den Wert ([value]) und eine Funktion, die bei jedem Tastenanschlag aufgerufen wird ([onChange]).

    • Zeilen 14–20: die Props; [help] ist optional;
    • Zeile 25: [useId()] liefert einen eindeutigen Bezeichner, der für jedes [<TextField>] auf der Seite verschieden ist, um [<label htmlFor>] und [<input id>] zu verknüpfen;
    • Zeile 31: Jeder Tastenanschlag ruft die Funktion der Elternkomponente mit dem neuen Text auf.

    Die Komponente hat keinen Zustand: Der Wert gehört der Elternkomponente, die den neuen Wert festlegt. Man spricht von „Lifting State Up“: Daten, die von mehreren Komponenten gemeinsam genutzt werden, liegen in ihrem nächsten gemeinsamen Vorfahren.

// 11-lifting-state-up/src/App.tsx - der Elternteil hört auf seine Kinder
// -------------------------------------------------------------------------
// Daten fließen nach unten (Props); Informationen fließen über
// in den Props übergebene Funktionen nach oben:
//   1. „Ereignisse“ des Kindes: onIncrement, onLimit;
//   2. eine kontrollierte Komponente: value + onChange, wie ein <input>. Der Zustand
//      bleibt Eigentum des Elternteils, der ihn selbst ändern kann (Clear).
// =========================================================================

import { useState } from 'react';
import Counter from './Counter';
import TextField from './TextField';

export default function App() {
  const [total, setTotal] = useState(0);
  const [message, setMessage] = useState('');
  const [firstName, setFirstName] = useState('Marie');
  const [lastName, setLastName] = useState('PELISSIER');

  // die von einem Zähler aufgerufene Funktion: Sie erhält den Wert, den er übergibt
  function onIncrement(label: string, value: number) {
    setTotal((t) => t + 1);
    setMessage(`${label} is now ${value}`);
  }

  function clear() {
    setFirstName('');
    setLastName('');
  }

  return (
    <div className="container py-4" style={{ maxWidth: 640 }}>
      <h1 className="h3 mb-3">Example 11 - the child notifies its parent</h1>

      <div className="card p-3 mb-4">
        <h2 className="h6">1. Functions called by the children</h2>
        <Counter label="Coffees" max={3} onIncrement={(v) => onIncrement('Coffees', v)}
                 onLimit={() => setMessage('No more than 3 coffees!')} />
        <Counter label="Glasses of water" max={8} onIncrement={(v) => onIncrement('Glasses of water', v)}
                 onLimit={() => setMessage('Enough drinking.')} />
        <p className="mb-0">Total of the clicks: {total} - <em>{message}</em></p>
      </div>

      <div className="card p-3">
        <h2 className="h6">2. Controlled components</h2>
        {/* onChange={setFirstName}: die Funktion des Zustands, unverändert übergeben */}
        <TextField value={firstName} onChange={setFirstName} label="First name" max={20} help="Required" />
        <TextField value={lastName} onChange={setLastName} label="Last name" max={20} help="Displayed in upper case" />
        <p className="mb-0">Data of the parent: <strong>{firstName} {lastName.toUpperCase()}</strong></p>
        <button className="btn btn-sm btn-outline-secondary mt-2 align-self-start" onClick={clear}>Clear (from the parent)</button>
      </div>
    </div>
  );
}
    • Zeilen 22–25: die Funktion, die von einem Zähler aufgerufen wird; sie erhält den Wert, den der Zähler ihr übergibt;
    • Zeilen 38–39: Die Elternkomponente übergibt Pfeilfunktionen, die das Label des Zählers ergänzen;
    • Zeile 48: [onChange={setFirstName}]: Die Zustandsfunktion wird unverändert übergeben; die Kindkomponente ruft sie mit dem neuen Text auf;
    • Zeile 51: Die Elternkomponente ändert die Werte selbst; die Felder folgen.

    // =========================================================================
// 12-composition/src/Card.tsx - ein wiederverwendbares TEMPLATE
// -------------------------------------------------------------------------
// Props übergeben DATEN... aber auch INHALT: JSX ist
// ein Wert wie jeder andere (vom Typ ReactNode), der an eine
// Komponente übergeben werden kann und den sie platziert, wo sie will:
//   - children: die besondere Prop, die erhält, was der Elternteil ZWISCHEN
//                <Card> und </Card> schreibt - den „Haupt“-Inhalt;
//   - header, footer: weitere Platzhalter, die der Elternteil in Form
//                von Props liefert; header hat einen Standardinhalt.
// Die Karte legt den Rahmen fest (Rand, Kopfbereich, Fußbereich), der Elternteil liefert, was
// darin steht. Es ist das clientseitige Gegenstück zum Template layout.hbs
// und seinem {{{body}}} im NestJS-Kurs.
// =========================================================================

import type { ReactNode } from 'react';

interface Props {
  color?: 'primary' | 'success' | 'danger';
  header?: ReactNode;
  footer?: ReactNode;
  children: ReactNode;
}

export default function Card({ color, header = 'Untitled', footer, children }: Props) {
  return (
    <div className={`card mb-3 ${color ? `border-${color}` : ''}`}>
      <div className={`card-header ${color ? `text-bg-${color}` : ''}`}>{header}</div>
      <div className="card-body">{children}</div>
      {/* kein Footer angegeben: überhaupt kein Footer */}
      {footer && <div className="card-footer text-muted small">{footer}</div>}
    </div>
  );
}
    • Zeilen 18–23: [header], [footer] und [children] sind [ReactNode]: JSX, Text, eine Zahl oder nichts;
    • Zeile 25: Standardinhalt;
    • Zeile 29: [children] ist eine besondere Property: Sie erhält alles, was die Elternkomponente ZWISCHEN [<Card>] und [</Card>] schreibt – den „Haupt“-Inhalt;
    • Zeile 31: kein Footer angegeben – also überhaupt kein Footer.

    Das Template legt den Rahmen fest (Rand, Header, Footer); die Elternkomponente liefert den Inhalt darin: Das entspricht der Inhaltsprojektion von [Angular] ([<ng-content>]) und den „Slots“ von [Vue.js] sowie, serverseitig, dem Template [layout.hbs] und seinem [{{{body}}}] im Kurs [NestJS].

        <tr>
          {columns.map((c) => <th key={c}>{c}</th>)}
        </tr>

          </tr>
        ) : (
          rows.map((l, i) => <tr key={l.id}>{renderRow(l, i)}</tr>)
        )}
      </tbody>
    </table>

}

// 13-lifecycle/src/Clock.tsx - eine Komponente, die startet und stoppt
// -------------------------------------------------------------------------
// Das Leben einer Komponente:
//   1. Sie wird ANGEZEIGT (gemountet): React ruft die Funktion auf und platziert ihr
//      Ergebnis in der Seite; dann führt es ihre Effekte aus (useEffect);
//   2. Sie wird AKTUALISIERT, so oft wie nötig (ihr Zustand oder ihre
//      Props ändern sich);
//   3. Sie wird aus der Seite ENTFERNT (unmountet): React führt das AUFRÄUMEN
//      (Cleanup) ihrer Effekte aus, die Funktion, die jeder Effekt zurückgegeben hat.
// Eine Uhr startet einen Timer (setInterval), wenn sie erscheint... und
// muss ihn unbedingt stoppen, wenn sie verschwindet: Sonst
// würde der Timer nutzlos weiterlaufen (Speicherleck).
// =========================================================================

import { useEffect, useEffectEvent, useState } from 'react';

interface Props {
  onLog: (message: string) => void;
}

export default function Clock({ onLog }: Props) {
  const [time, setTime] = useState(() => new Date().toLocaleTimeString());

  // useEffectEvent: eine Funktion, die VON einem Effekt aufgerufen wird und stets
  // die aktuellsten Props liest, ohne eine Abhängigkeit des Effekts zu sein. Ohne
  // sie müsste onLog unten in die Abhängigkeiten [...] aufgenommen werden,
  // und da der Elternteil bei jeder Anzeige eine NEUE Funktion übergibt, würde der
  // Timer jede Sekunde gestoppt und neu gestartet
  const log = useEffectEvent((message: string) => onLog(message));

  // []: keine Abhängigkeit. Der Effekt läuft EINMAL, nach der ersten
  // Anzeige; die Funktion, die er zurückgibt, EINMAL, wenn die Komponente entfernt wird
  useEffect(() => {
    log('Clock: effect - starting the timer');
    const timer = setInterval(() => setTime(new Date().toLocaleTimeString()), 1000);
    // das AUFRÄUMEN (Cleanup)
    return () => {
      clearInterval(timer);
      log('Clock: cleanup - timer stopped');
    };
  }, []);

  return <div className="display-6 font-monospace">{time}</div>;
}

    Der Lebenszyklus einer Komponente (Zeilen 4–10): Sie wird gemountet (zum ersten Mal angezeigt), aktualisiert so oft wie nötig und schließlich unmountet (von der Seite entfernt). [React] kennt keine eigenen Methoden für die einzelnen Phasen: Alles läuft über [useEffect].

    • Zeilen 34–42: [useEffect(effect, [])]: Bei einem LEEREN Abhängigkeits-Array wird der Effekt nur einmal ausgeführt, nach der ersten Anzeige. Die Funktion, die er zurückgibt (Zeilen 38–41), übernimmt die Bereinigung: [React] führt sie aus, wenn die Komponente von der Seite entfernt wird (und vor jeder erneuten Ausführung des Effekts, wenn sich seine Abhängigkeiten ändern). Eine Uhr startet beim Start einen Timer ([setInterval]) ... und muss ihn beenden ([clearInterval]), wenn sie verschwindet: Andernfalls würde der Timer grundlos weiterlaufen und versuchen, den Zustand einer Komponente zu ändern, die nicht mehr existiert;
    • Zeile 30: Der Effekt ruft die Funktion [onLog] der Elternkomponente auf. Streng genommen müsste er sie in seine Abhängigkeiten aufnehmen: Da der Effekt eine Prop verwendet, muss er erneut ausgelöst werden, wenn sich die Prop ändert. Die Elternkomponente erzeugt jedoch bei jeder Anzeige eine NEUE Funktion: Der Timer würde ständig gestoppt und neu gestartet. [useEffectEvent] ([React] 19.2) erzeugt eine Funktion, die immer die jeweils aktuellste Version von [onLog] aufruft, ohne eine Abhängigkeit des Effekts zu sein. Sie wird nur innerhalb eines Effekts aufgerufen.

// 14-contexts/src/contexts/log.tsx - ein Kontext für die GESAMTE Anwendung
// -------------------------------------------------------------------------
// Wie teilt man Daten und Funktionen zwischen entfernten Komponenten,
// ohne sie von Props zu Props durch alle
// Zwischenstationen zu reichen? Mit einem KONTEXT:
//   1. createContext(defaultValue) erstellt den Kontext;
//   2. eine PROVIDER-Komponente, <LogContext value={...}>, gibt ihren
//      Wert an ALLE ihre Nachkommen weiter, in beliebiger Tiefe;
//   3. ein Nachkomme LIEST ihn mit use(LogContext) (oder useContext).
// Wenn sich der bereitgestellte Wert ändert, werden alle Komponenten, die ihn lesen,
// erneut aufgerufen.
//
// Die Konvention: eine Datei pro Kontext, die den Provider (eine
// Komponente, die den Zustand hält) und einen HOOK zum Lesen (useLog) exportiert.
// So teilt React, was Angular in seinen Services ablegt.
// =========================================================================

import { createContext, use, useCallback, useMemo, useState, type ReactNode } from 'react';

interface Log {
  lines: string[];
  trace: (message: string) => void;
}

// null: der gelesene Wert, wenn KEIN Provider die Komponente umschließt
const LogContext = createContext<Log | null>(null);

// der Provider: Er hält den Zustand und stellt ihn seinen Nachkommen bereit
export function LogProvider({ children }: { children: ReactNode }) {
  const [lines, setLines] = useState<string[]>([]);

  // useCallback: dieselbe Funktion von einer Anzeige zur nächsten (und nicht
  // jedes Mal eine neue), damit die Komponenten, die sie erhalten,
  // nicht verändert erscheinen
  const trace = useCallback((message: string) => {
    setLines((l) => [`${new Date().toLocaleTimeString()} ${message}`, ...l].slice(0, 6));
  }, []);

  // useMemo: nur dann ein neues Objekt, wenn sich lines geändert hat
  const value = useMemo(() => ({ lines, trace }), [lines, trace]);

  // React 19: der Kontext selbst ist die Provider-Komponente
  return <LogContext value={value}>{children}</LogContext>;
}

// der Lese-Hook: ein Name, der mit „use“ beginnt
export function useLog(): Log {
  // use(context): der Wert des NÄCHSTGELEGENEN Providers
  const log = use(LogContext);
  if (log === null) {
    throw new Error('useLog must be used inside a <LogProvider>');
  }
  return log;
}

// -------------------------------------------------------------------------
// Ein EIGENER HOOK (Custom Hook) ist eine Funktion, deren Name mit „use“ beginnt und
// die selbst Hooks aufruft (useState, useEffect, useRef...). Er
// erlaubt es, ein VERHALTEN (mit seinem Zustand) in mehreren

import { useEffect, useRef, useState } from 'react';

export function useLongPress(action: () => void, duration = 1000) {
  const [pressing, setPressing] = useState(false);
  // ein Wert, der die Anzeigen überdauert, OHNE eine auszulösen: useRef
  const timer = useRef<number | undefined>(undefined);

  // wenn die Komponente entfernt wird: kein verwaister Timer
  useEffect(() => () => clearTimeout(timer.current), []);

  function start() {
    setPressing(true);
    timer.current = window.setTimeout(() => {
      setPressing(false);
      action();
    }, duration);
  }

  function stop() {
    clearTimeout(timer.current);
    setPressing(false);
  }

  return {
    pressing,
    handlers: { onPointerDown: start, onPointerUp: stop, onPointerLeave: stop },
  };
}

// "Are you sure?": window.confirm() gibt es zwar, aber es lässt sich weder
// gestalten, noch lassen sich seine Schaltflächen übersetzen. Es wird durch einen

// oder false (Nein), sobald der Benutzer antwortet. Der aufrufende Code wird dann

import { createContext, use, useCallback, useRef, useState, type ReactNode } from 'react';
import ConfirmDialog from './ConfirmDialog';

type Confirm = (message: string) => Promise<boolean>;

const ConfirmationContext = createContext<Confirm | null>(null);

export function ConfirmationProvider({ children }: { children: ReactNode }) {
  const [message, setMessage] = useState<string | null>(null); // null: Dialog geschlossen
  // die Funktion, die das aktuelle Promise auflöst: Sie wird nicht angezeigt,
  // ist daher eine Referenz und kein Zustand
  const answer = useRef<((yes: boolean) => void) | null>(null);

  const confirm = useCallback<Confirm>((text) => {
    answer.current?.(false); // eine noch offene Frage? Das bedeutet „Nein“
    setMessage(text);
    return new Promise<boolean>((resolve) => {
      answer.current = resolve;
    });
  }, []);

  function respond(yes: boolean) {
    setMessage(null);
    const r = answer.current;
    answer.current = null;
    r?.(yes);
  }

  return (
    <ConfirmationContext value={confirm}>
      {children}
      {message !== null && <ConfirmDialog message={message} onAnswer={respond} />}
    </ConfirmationContext>
  );
}

export function useConfirmation(): Confirm {
  const confirm = use(ConfirmationContext);
  if (confirm === null) {
    throw new Error('useConfirmation must be used inside a <ConfirmationProvider>');
  }
  return confirm;
}

    Tipp: [confirm(message)] gibt ein Promise zurück, das mit [true] (Ja) oder [false] (Nein) aufgelöst wird, sobald der Benutzer antwortet. Der aufrufende Code ist dann genauso einfach wie mit [window.confirm]: [if (await confirm('Delete this doctor?')) {...}].

  message: string;
  onAnswer: (yes: boolean) => void;
}

  // und beim Schließen entfernt wird (Cleanup des Effekts)
  useEffect(() => {
    const onKeydown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onAnswer(false);
    };
    document.addEventListener('keydown', onKeydown);
    return () => document.removeEventListener('keydown', onKeydown);
  }, [onAnswer]);

  return createPortal(
    <>
      <div className="modal-backdrop fade show"></div>
      {/* ein Klick auf den Hintergrund (und nicht in den Dialog) bedeutet „Nein“ */}
      <div className="modal fade show d-block" tabIndex={-1} role="alertdialog" aria-modal="true"
           onClick={(e) => e.target === e.currentTarget && onAnswer(false)}>
        <div className="modal-dialog modal-dialog-centered">
          <div className="modal-content">
            <div className="modal-header">
              <h5 className="modal-title">Confirmation</h5>
              <button type="button" className="btn-close" aria-label="Close" onClick={() => onAnswer(false)}></button>
            </div>
            <div className="modal-body">
              <p className="mb-0">{message}</p>
            </div>
            <div className="modal-footer">
              {/* autoFocus: React gibt dieser Schaltfläche den Fokus, wenn sie erscheint */}
              <button type="button" className="btn btn-secondary" autoFocus onClick={() => onAnswer(false)}>No, go back</button>
              <button type="button" className="btn btn-danger" onClick={() => onAnswer(true)}>Yes, confirm</button>
            </div>
          </div>
        </div>
      </div>
    </>,
    document.body,
  );
}

import { useState } from 'react';
import { useConfirmation } from './confirmation/confirmation';

interface Contact {
  id: number;
  name: string;
}

export default function App() {
  const confirm = useConfirmation();
  const [contacts, setContacts] = useState<Contact[]>([
    { id: 1, name: 'Jean MARTIN' },
    { id: 2, name: 'Alice DURAND' },
    { id: 3, name: 'Luc GERMAN' },
  ]);
  const [message, setMessage] = useState('');

  // eine async-Funktion: Sie „wartet“ auf die Antwort des Benutzers
  async function remove(c: Contact) {
    if (!(await confirm(`Delete the contact ${c.name}?`))) {
      setMessage('Deletion cancelled.');
      return;
    }
    setContacts((list) => list.filter((x) => x.id !== c.id));
    setMessage(`${c.name} has been deleted.`);
  }

  async function removeAll() {
    if (await confirm(`Delete the ${contacts.length} contacts?`)) {
      setContacts([]);
      setMessage('All the contacts have been deleted.');
    }
  }

  return (
    <div className="container py-4" style={{ maxWidth: 640 }}>
      <h1 className="h3 mb-3">Example 16 - a confirmation dialog</h1>
      <ul className="list-group mb-3">
        {contacts.map((c) => (
          <li key={c.id} className="list-group-item d-flex justify-content-between align-items-center">
            {c.name}
            <button className="btn btn-sm btn-outline-danger" onClick={() => remove(c)}>Delete</button>
          </li>
        ))}
        {contacts.length === 0 && <li className="list-group-item text-muted">No contacts.</li>}
      </ul>
      <button className="btn btn-danger" disabled={contacts.length === 0} onClick={removeAll}>Delete all</button>
      {message && <p className="alert alert-info mt-3">{message}</p>}
    </div>
  );
}
